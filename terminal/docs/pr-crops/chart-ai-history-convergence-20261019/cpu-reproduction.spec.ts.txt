import { expect, test } from "@playwright/test";
import { isolateWatchlistStore } from "./watchlistStore";

// Temporary local diagnostic. Uses only the incumbent isolated fixture store.
const query = "mo_from=transmission&mo_chain=context-chain&mo_focus=leg-3&mo_path_rev=12&mo_asof=2026-10-01&mo_security=SEC:US";
for (const rate of [4, 8]) {
  test(`history diagnostic CPU ${rate}`, async ({ page, baseURL }, testInfo) => {
    await isolateWatchlistStore(page, testInfo, baseURL);
    await page.addInitScript(() => { localStorage.setItem("mm.lang", "en"); });
    const response = await page.request.post("/api/theses", { data: {
      action: "create", clientRequestId: "a1000000-0000-4000-8000-00000000c3de",
      subject: {
        schema: "mastermind.thesis-subject-ref/v1", kind: "issuer", owner: "terminal.analysis_symbol",
        key: "NVDA", identityState: "listing_scoped", listing: { symbol: "NVDA", mic: null, securityId: null },
        companyId: null, display: "NVDA · listing scoped",
      },
      content: {
        schema: "mastermind.thesis-content/v1", title: "Rapid history thesis",
        statement: "This existing thesis is reopened while a MarketOntology context is on screen.",
        catalysts: [], falsifiers: [], risks: [], horizon: "unspecified", effectiveAt: null, revisionNote: null,
      },
    } });
    expect(response.ok()).toBe(true);
    await page.goto(`/analysis?view=theses&symbol=NVDA&${query}`);
    await page.getByRole("button", { name: "Rapid history thesis" }).click();
    await expect(page.getByText("Version 1 · Current")).toBeVisible();
    await expect(page.getByTestId("mo-context-strip")).toBeVisible();
    const selectedUrl = page.url();
    const cdp = await page.context().newCDPSession(page);
    await cdp.send("Emulation.setCPUThrottlingRate", { rate });
    await page.reload();
    await expect(page.getByText("Version 1 · Current")).toBeVisible();
    await page.goBack();
    await page.goForward();
    await expect(page).toHaveURL(selectedUrl);
    await expect(page.getByText("Version 1 · Current")).toBeVisible();
    await expect(page.getByTestId("mo-context-strip")).toBeVisible();
  });
}
